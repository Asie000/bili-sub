> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现一个原生 Windows 上的 Go 工具，能从单视频/UP主空间/收藏夹批量探测并下载 B站字幕，可选本地 Whisper 转写，导出为 zip，并提供浏览器界面。

**Architecture:** Go 控制面（net/http 客户端 + WBI 签名 + 并发 Worker + rate.Limiter + SQLite 状态机）与本地 ASR 算力面（Python 常驻 worker）解耦；所有字幕先转 Canonical IR，导出为纯内存渲染；Web 界面由 Go 内嵌静态页 + JSON API + SSE 提供。

**Tech Stack:** Go 1.27（标准库 net/http、database/sql）、modernc.org/sqlite、golang.org/x/time/rate、spf13/cobra、Vue 无关的原生 HTML/JS、faster-whisper（Python 3.11/3.12 venv）、ffmpeg。

**Spec:** `docs/superpowers/specs/2026-10-08-bili-sub-design.md`

package config

import (
	"os"
	"path/filepath"
	"strconv"
	"strings"
)

type Config struct {
	SESSDATA    string
	DataDir     string
	DBPath      string
	Concurrency int
	QPS         float64
	FFmpegPath  string
	PythonPath  string
}

func Default() *Config {
	dataDir := "data"
	return &Config{
		DataDir:     dataDir,
		DBPath:      filepath.Join(dataDir, "bili_sub.db"),
		Concurrency: 3,
		QPS:         2,
		FFmpegPath:  "ffmpeg",
		PythonPath:  "python",
	}
}

// Load 读取 .env 风格文件（KEY=VALUE），缺失则用默认值。
func Load(envPath string) (*Config, error) {
	c := Default()
	if envPath == "" {
		envPath = ".env"
	}
	b, err := os.ReadFile(envPath)
	if err != nil {
		if os.IsNotExist(err) {
			c.applyEnv()
			return c, nil
		}
		return nil, err
	}
	for _, line := range strings.Split(string(b), "\n") {
		line = strings.TrimSpace(line)
		if line == "" || strings.HasPrefix(line, "#") || !strings.Contains(line, "=") {
			continue
		}
		kv := strings.SplitN(line, "=", 2)
		c.set(strings.TrimSpace(kv[0]), strings.TrimSpace(kv[1]))
	}
	c.applyEnv()
	return c, nil
}

func (c *Config) applyEnv() {
	for _, k := range []string{"SESSDATA", "BILI_CONCURRENCY", "BILI_QPS", "FFMPEG_PATH", "PYTHON_PATH", "BILI_DATA_DIR"} {
		if v := os.Getenv(k); v != "" {
			c.set(k, v)
		}
	}
}

func (c *Config) set(k, v string) {
	switch k {
	case "SESSDATA":
		c.SESSDATA = v
	case "BILI_DATA_DIR":
		c.DataDir = v
		c.DBPath = filepath.Join(v, "bili_sub.db")
	case "BILI_CONCURRENCY":
		if n, err := strconv.Atoi(v); err == nil {
			c.Concurrency = n
		}
	case "BILI_QPS":
		if f, err := strconv.ParseFloat(v, 64); err == nil {
			c.QPS = f
		}
	case "FFMPEG_PATH":
		c.FFmpegPath = v
	case "PYTHON_PATH":
		c.PythonPath = v
	}
}
```

package ir

import "strings"

type Segment struct {
	StartMs int64  `json:"start_ms"`
	EndMs   int64  `json:"end_ms"`
	Text    string `json:"text"`
	Speaker string `json:"speaker,omitempty"`
	HasTime bool   `json:"-"` // 导出时是否保留时间轴由渲染器决定
}

type Document struct {
	Bvid     string    `json:"bvid"`
	Cid      int64     `json:"cid"`
	Title    string    `json:"title"`
	Author   string    `json:"author"`
	Lang     string    `json:"lang"`
	Segments []Segment `json:"segments"`
}

func (d *Document) PlainText() string {
	parts := make([]string, 0, len(d.Segments))
	for _, s := range d.Segments {
		t := strings.TrimSpace(s.Text)
		if t != "" {
			parts = append(parts, t)
		}
	}
	return strings.Join(parts, "\n")
}

func (d *Document) DurationMs() int64 {
	var max int64
	for _, s := range d.Segments {
		if s.EndMs > max {
			max = s.EndMs
		}
	}
	return max
}
```

package util

import "strings"

var badChars = strings.NewReplacer(
	`\`, "_", `/`, "_", `:`, "_", `*`, "_", `?`, "_",
	`"`, "_", `<`, "_", `>`, "_", `|`, "_",
	"\r", "", "\n", " ", "\t", " ",
)

func SanitizeFilename(title, bvid string) string {
	s := badChars.Replace(title)
	s = strings.TrimSpace(s)
	r := []rune(s)
	const maxTitle = 100
	if len(r) > maxTitle {
		r = r[:maxTitle]
		s = string(r)
	}
	if s == "" {
		s = "untitled"
	}
	return s + " [" + bvid + "]"
}

```go
package export

import (
	"bytes"
	"encoding/json"
	"fmt"
	"strings"

	"bilibili-sub/internal/ir"
)

const (
	FormatSRT      = "srt"
	FormatText     = "txt"
	FormatMarkdown = "md"
	FormatJSON     = "json"
)

func Render(d *ir.Document, format string) ([]byte, error) {
	switch format {
	case FormatSRT:
		return renderSRT(d), nil
	case FormatText:
		return []byte(d.PlainText()), nil
	case FormatMarkdown:
		return renderMD(d), nil
	case FormatJSON:
		return json.MarshalIndent(d, "", "  ")
	default:
		return nil, fmt.Errorf("unsupported format: %s", format)
	}
}

func ts(ms int64) string {
	h := ms / 3600000
	m := (ms % 3600000) / 60000
	s := (ms % 60000) / 1000
	mmm := ms % 1000
	return fmt.Sprintf("%02d:%02d:%02d,%03d", h, m, s, mmm)
}

func renderSRT(d *ir.Document) []byte {
	var b bytes.Buffer
	for i, seg := range d.Segments {
		if strings.TrimSpace(seg.Text) == "" {
			continue
		}
		fmt.Fprintf(&b, "%d\n%s --> %s\n%s\n\n", i+1, ts(seg.StartMs), ts(seg.EndMs), strings.TrimSpace(seg.Text))
	}
	return b.Bytes()
}

func renderMD(d *ir.Document) []byte {
	var b bytes.Buffer
	title := d.Title
	if title == "" {
		title = d.Bvid
	}
	fmt.Fprintf(&b, "# %s\n\n", title)
	if d.Author != "" {
		fmt.Fprintf(&b, "- 作者：%s\n", d.Author)
	}
	fmt.Fprintf(&b, "- 视频：%s\n\n", d.Bvid)
	for _, seg := range d.Segments {
		t := strings.TrimSpace(seg.Text)
		if t == "" {
			continue
		}
		fmt.Fprintf(&b, "%s\n\n", t)
	}
	return b.Bytes()
}
```

`internal/bili/wbi.go`：
```go
package bili

import (
	"crypto/md5"
	"encoding/hex"
	"fmt"
	"net/url"
	"sort"
	"strconv"
	"strings"
	"time"
)

var mixinKeyEncTab = []int{
	46, 47, 18, 2, 53, 8, 23, 32, 15, 50, 10, 31, 58, 3, 45, 35, 27, 43, 5, 49,
	33, 9, 42, 19, 29, 28, 14, 39, 12, 38, 41, 13, 37, 48, 7, 16, 24, 55, 40,
	61, 26, 17, 0, 1, 60, 51, 30, 4, 22, 25, 54, 21, 56, 59, 6, 63, 57, 62, 11,
	36, 20, 34, 44, 52,
}

type WBISigner struct {
	mixinKey string
}

func NewWBISigner(imgKey, subKey string) (*WBISigner, error) {
	orig := imgKey + subKey
	if len(orig) < 64 {
		return nil, fmt.Errorf("wbi key too short: %d", len(orig))
	}
	var sb strings.Builder
	for _, i := range mixinKeyEncTab {
		sb.WriteByte(orig[i])
	}
	return &WBISigner{mixinKey: sb.String()[:32]}, nil
}

func (s *WBISigner) Sign(params map[string]string) (map[string]string, error) {
	out := make(map[string]string, len(params)+2)
	for k, v := range params {
		out[k] = v
	}
	if _, ok := out["wts"]; !ok {
		out["wts"] = strconv.FormatInt(time.Now().Unix(), 10)
	}
	keys := make([]string, 0, len(out))
	for k := range out {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	var q strings.Builder
	for i, k := range keys {
		if i > 0 {
			q.WriteByte('&')
		}
		q.WriteString(url.QueryEscape(k))
		q.WriteByte('=')
		q.WriteString(url.QueryEscape(out[k]))
	}
	sum := md5.Sum([]byte(q.String() + s.mixinKey))
	out["w_rid"] = hex.EncodeToString(sum[:])
	return out, nil
}
```

package bili

import (
	"context"
	"encoding/json"
	"fmt"
	"net/http"
	"net/url"
	"time"
)

const (
	userAgent = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0 Safari/537.36"
	referer   = "https://www.bilibili.com"
)

type Limiter interface {
	Wait(ctx context.Context) error
}

type APIError struct {
	Code    int
	Message string
}

func (e *APIError) Error() string { return fmt.Sprintf("bilibili code=%d message=%s", e.Code, e.Message) }
func (e *APIError) Unauthorized() bool { return e.Code == -101 || e.Code == -400 }

type Client struct {
	HTTP     *http.Client
	SESSDATA string
	Limiter  Limiter
}

func NewClient(sessdata string, limiter Limiter) *Client {
	return &Client{
		HTTP:     &http.Client{Timeout: 20 * time.Second},
		SESSDATA: sessdata,
		Limiter:  limiter,
	}
}

func (c *Client) Get(ctx context.Context, rawURL string, params map[string]string, out any) error {
	if c.Limiter != nil {
		if err := c.Limiter.Wait(ctx); err != nil {
			return err
		}
	}
	u, err := url.Parse(rawURL)
	if err != nil {
		return err
	}
	q := u.Query()
	for k, v := range params {
		q.Set(k, v)
	}
	u.RawQuery = q.Encode()

	req, err := http.NewRequestWithContext(ctx, http.MethodGet, u.String(), nil)
	if err != nil {
		return err
	}
	req.Header.Set("User-Agent", userAgent)
	req.Header.Set("Referer", referer)
	if c.SESSDATA != "" {
		req.Header.Set("Cookie", "SESSDATA="+c.SESSDATA)
	}
	resp, err := c.HTTP.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return fmt.Errorf("HTTP %d", resp.StatusCode)
	}
	var env struct {
		Code    int             `json:"code"`
		Message string          `json:"message"`
		Data    json.RawMessage `json:"data"`
	}
	if err := json.NewDecoder(resp.Body).Decode(&env); err != nil {
		return err
	}
	if env.Code != 0 {
		return &APIError{Code: env.Code, Message: env.Message}
	}
	if out != nil && len(env.Data) > 0 && string(env.Data) != "null" {
		return json.Unmarshal(env.Data, out)
	}
	return nil
}

package bili

import (
	"context"
	"math/rand"
	"time"

	"golang.org/x/time/rate"
)

type rateLimiter struct {
	lim *rate.Limiter
}

func NewRateLimiter(qps float64) Limiter {
	if qps <= 0 {
		qps = 2
	}
	return &rateLimiter{lim: rate.NewLimiter(rate.Limit(qps), 1)}
}

func (r *rateLimiter) Wait(ctx context.Context) error {
	if err := r.lim.Wait(ctx); err != nil {
		return err
	}
	// 抖动，避免节奏过于规律被风控
	jitter := time.Duration(rand.Int63n(int64(200 * time.Millisecond)))
	select {
	case <-time.After(jitter):
	case <-ctx.Done():
		return ctx.Err()
	}
	return nil
}

		VALUES (?,?,?,?,?,?,?,?,?,?,?,?)
		ON CONFLICT(bvid,cid) DO UPDATE SET
			author=excluded.author, title=excluded.title, pubdate=excluded.pubdate,
			duration=excluded.duration, page_index=excluded.page_index, part=excluded.part,
